SIGNALS IN:
  4H:
    DVG4..DVG7   -- vector data bus bits 4-7
    /LATCH1      -- positive-edge clock input
    P,R110       -- 1K pull-up to +5 V; holds active-low clear inactive
  3C:
    DVG4..DVG7   -- vector data bus bits 4-7
    /LATCH3      -- positive-edge clock input
    /LATCH1      -- active-low asynchronous clear

SIGNALS OUT:
  4H:
    OP2, /OP2
    OP1, /OP1
    OP0, /OP0
    DVY12, /DVY12
  3C:
    Z2, /Z2
    Z1, /Z1
    Z0, /Z0
    DVX12, /DVX12

CHIPS:
  4H    LS175    -- opcode and DVY12 latch
  3C    LS175    -- intensity and DVX12 latch

LOGIC:
  4H(LS175):     CK = /LATCH1                       -- pin 9
                 /CLR = P,R110                      -- pin 1, pulled HIGH
                 DVG7 -> OP2,   /OP2                -- pins 4 -> 2,3
                 DVG6 -> OP1,   /OP1                -- pins 12 -> 10,11
                 DVG5 -> OP0,   /OP0                -- pins 5 -> 7,6
                 DVG4 -> DVY12, /DVY12              -- pins 13 -> 15,14

  3C(LS175):     CK = /LATCH3                       -- pin 9
                 /CLR = /LATCH1                     -- pin 1
                 DVG7 -> Z2,    /Z2                 -- pins 12 -> 10,11
                 DVG6 -> Z1,    /Z1                 -- pins 5 -> 7,6
                 DVG5 -> Z0,    /Z0                 -- pins 13 -> 15,14
                 DVG4 -> DVX12, /DVX12              -- pins 4 -> 2,3

NOTES:
  - P,R110 is the R110 1K pull-up to +5 V shown on Sheet 11B,
    PDF page 22.
  - /LATCH1 asynchronously clears 3C whenever it is LOW. Its rising edge also
    clocks 4H.

SIGNALS IN:
  DVG0..DVG7     -- parallel-load data
  /ENORM         -- active-low normalization-shift enable
  /LATCH0        -- load DVY bits 0-7
  /LATCH1        -- load DVY bits 8-11; also clears five shifters and 3C
  /LATCH2        -- load DVX bits 0-7
  /LATCH3        -- load DVX bits 8-11
  P,R110         -- 1K pull-up to +5 V
  12 MHZ         -- shifter clock

SIGNALS OUT:
  DVY0..DVY11
  DVX0..DVX11

CHIPS:
  5F    LS194    -- DVY bits 0-3;  /CLR = /LATCH1
  5H    LS194    -- DVY bits 4-7;  /CLR = /LATCH1
  6H    LS194    -- DVY bits 8-11; /CLR = P,R110
  5A    LS194    -- DVX bits 0-3;  /CLR = /LATCH1
  5B    LS194    -- DVX bits 4-7;  /CLR = /LATCH1
  5C    LS194    -- DVX bits 8-11; /CLR = /LATCH1
  4E    S00      -- quad 2-input NAND; drives LS194 S1
  4D    LS14     -- hex inverter; drives LS194 S0

MODE CONTROL:
  4E(S00):
    pins 1=/LATCH0,  2=/ENORM  -> pin 3  -> 5F/5H S1(10)
    pins 4=/LATCH1,  5=/ENORM  -> pin 6  -> 6H S1(10)
    pins 9=/LATCH2, 10=/ENORM  -> pin 8  -> 5A/5B S1(10)
    pins 13=/LATCH3,12=/ENORM  -> pin 11 -> 5C S1(10)

  4D(LS14):
    pin 1=/LATCH0  -> pin 2  -> 5F/5H S0(9)
    pin 3=/LATCH1  -> pin 4  -> 6H S0(9)
    pin 5=/LATCH2  -> pin 6  -> 5A/5B S0(9)
    pin 13=/LATCH3 -> pin 12 -> 5C S0(9)

  Resulting LS194 modes:
    no latch, /ENORM=1 -> S1:S0 = 00 -> hold
    no latch, /ENORM=0 -> S1:S0 = 10 -> shift left
    selected latch LOW -> S1:S0 = 11 -> parallel load

DVY CHAIN:
  5F(LS194):
    /CLR(1) = /LATCH1
    SR(2)   = P,R110
    SL(7)   = GND
    S0(9)   = 4D pin 2
    S1(10)  = 4E pin 3
    CK(11)  = 12 MHZ
    D,C,B,A = DVG0,DVG1,DVG2,DVG3         -- pins 6,5,4,3
    QD,QC,QB,QA = DVY0,DVY1,DVY2,DVY3    -- pins 12,13,14,15

  5H(LS194):
    /CLR(1) = /LATCH1
    SR(2)   = P,R110
    SL(7)   = DVY3                        -- from 5F QA pin 15
    S0(9)   = 4D pin 2
    S1(10)  = 4E pin 3
    CK(11)  = 12 MHZ
    QD,QC,QB,QA = DVY4,DVY5,DVY6,DVY7

  6H(LS194):
    /CLR(1) = P,R110                      -- held inactive HIGH
    SR(2)   = P,R110
    SL(7)   = DVY7                        -- from 5H QA pin 15
    S0(9)   = 4D pin 4
    S1(10)  = 4E pin 6
    CK(11)  = 12 MHZ
    QD,QC,QB,QA = DVY8,DVY9,DVY10,DVY11

DVX CHAIN:
  5A(LS194):
    /CLR(1) = /LATCH1
    SR(2)   = P,R110
    SL(7)   = GND
    S0(9)   = 4D pin 6
    S1(10)  = 4E pin 8
    CK(11)  = 12 MHZ
    D,C,B,A = DVG0,DVG1,DVG2,DVG3
    QD,QC,QB,QA = DVX0,DVX1,DVX2,DVX3

  5B(LS194):
    /CLR(1) = /LATCH1
    SR(2)   = P,R110
    SL(7)   = DVX3                        -- from 5A QA pin 15
    S0(9)   = 4D pin 6
    S1(10)  = 4E pin 8
    CK(11)  = 12 MHZ
    QD,QC,QB,QA = DVX4,DVX5,DVX6,DVX7

  5C(LS194):
    /CLR(1) = /LATCH1
    SR(2)   = P,R110
    SL(7)   = DVX7                        -- from 5B QA pin 15
    S0(9)   = 4D pin 12
    S1(10)  = 4E pin 11
    CK(11)  = 12 MHZ
    QD,QC,QB,QA = DVX8,DVX9,DVX10,DVX11

NOTES:
  - Each normalization cascade connection runs from QA pin 15 of one LS194
    to SL pin 7 of the next LS194.
  - With DVX/DVY bit 0 at QD and the LS194 in shift-left mode, each shift is
    a numerical left shift. Zero enters bit 0 through the first SL input.
  - P,R110 holds the active-low clear of 6H HIGH.

; ------------------------------------------------------------------------------
; BLOCK: Normalization Flag               *** VERIFIED 2026-07-16 ***
; Sheet 13A, PDF page 25
; ------------------------------------------------------------------------------
;
; Generates the active-low shift control for the LS194 vector-data shifters.

SIGNALS IN:
  DVY12          -- 4H LS175 Q, vector Y sign bit
  DVY11          -- 6H LS194 QA, vector Y data MSB
  DVX12          -- 3C LS175 Q, vector X sign bit
  DVX11          -- 5C LS194 QA, vector X data MSB
  VCTR           -- vector draw in progress
  SCALE          -- vector scaling in progress
  CNTR           -- center operation in progress
  /OP0           -- 4H LS175 inverted opcode bit 0
  /STROBE0       -- active-low state-machine strobe
  P,R109         -- 1K pull-up to +5 V
  /SA            -- active-low signature-analysis test input

SIGNALS OUT:
  NORM           -- normalization flag
  /ENORM         -- active-low LS194 shift enable

CHIPS:
  4F    S86      -- quad 2-input XOR; two gates used
  3F    S260     -- dual 5-input NOR; one gate used
  5K    S74      -- D flip-flop
  6J    S10      -- triple 3-input NAND; one gate used

LOGIC:
  4F(S86):
    DVY_DIFF = DVY12 XOR DVY11                  -- pins 2,1 -> 3
    DVX_DIFF = DVX12 XOR DVX11                  -- pins 9,10 -> 8

  3F(S260):
    NORM_CLEAR_N = NOR5(DVY_DIFF,                -- pin 1
                        DVX_DIFF,                -- pin 2
                        VCTR,                    -- pin 3
                        SCALE,                   -- pin 12
                        CNTR)                    -- pin 13
    output = NORM_CLEAR_N                        -- pin 5

  5K(S74):
    D      = /OP0                               -- pin 12
    CK     = /STROBE0                           -- pin 11
    /PRE   = P,R109                             -- pin 10, pulled HIGH
    /CLR   = NORM_CLEAR_N                       -- pin 13
    Q      = NORM                               -- pin 9

  6J(S10):
    /ENORM = NAND(NORM, NORM_CLEAR_N, /SA)      -- pins 9,10,11 -> 8

NOTES:
  - The rising edge at the end of /STROBE0 clocks /OP0 into NORM.
  - NORM_CLEAR_N stays HIGH only while both sign comparisons match and VCTR,
    SCALE, and CNTR are all LOW. Any listed termination input asynchronously
    clears NORM and returns /ENORM HIGH.
  - NORM controls the LS194 shift mode through /ENORM and drives the Vector
    Timer load-control logic.
  - Grounding the SA test point makes /SA LOW and forces /ENORM HIGH, disabling
    normalization shifts during signature analysis.

SIGNALS IN:
  OP0            -- opcode bit 0
  /STROBE3       -- active-low state-machine strobe
  /VGRST         -- active-low vector-generator reset
  /VGGO          -- active-low vector-generator start

SIGNALS OUT:
  HALT
  /HALT

CHIPS:
  1H    LS74     -- D flip-flop

LOGIC:
  1H(LS74):      D    = OP0                         -- pin 12
                 CK   = /STROBE3                    -- pin 11
                 /PRE = /VGRST                      -- pin 10
                 /CLR = /VGGO                       -- pin 13
                 Q    = HALT                        -- pin 9
                 /Q   = /HALT                       -- pin 8

NOTES:
  - /VGRST presets HALT; /VGGO clears it.
  - HALT is sampled into 3B on the state-machine clock to produce HALT*.

; BLOCK: Signature Analysis               *** VERIFIED 2026-07-16 ***
; Sheet 13A, PDF page 25

SIGNALS IN:
  /SA            -- active-low diagnostic input at the SA test point

SIGNALS OUT:
  SA             -- active-high signature-analysis selection
  /SA            -- active-low signature-analysis selection

CHIPS:
  6K    LS04     -- hex inverter; one gate used
  R10   10K      -- pull-up resistor to +5 V

LOGIC:
  R10:           +5 V -> /SA test-point node
  6K(LS04):      SA = NOT(/SA)                     -- pins 9 -> 8

DESTINATIONS:
  SA:
    4C LS157 select pin 1
    4A LS157 select pin 1
    2E LS32 pin 13 in Decoder Disable
    5J LS32 pin 4 in the Address Controller

  /SA:
    6J S10 pin 11 in the Normalization Flag
    3A LS00 pin 4 in the Vector Timer
    2A LS20 pin 12 in the Vector Timer
    2H LS08 pin 1 in the Address Controller

NOTES:
  - In normal operation R10 holds /SA HIGH and SA LOW.
  - Signature-analysis mode is selected by grounding the SA test point. This
    makes /SA LOW and SA HIGH.
  - SA selects vector-address lines AM7..AM0 at the state PROM.


; ------------------------------------------------------------------------------
; BLOCK: State Machine (PROM and Control) *** VERIFIED 2026-07-16 ***
; Sheet 13A, PDF page 25
; ------------------------------------------------------------------------------

SIGNALS IN:
  GO             -- VCTR OR CNTR
  HALT           -- raw halt flag, sampled by 3B
  HALT*          -- clocked halt feedback from 3B to 1E
  OP2,OP1,OP0    -- opcode bits
  SA             -- signature-analysis mux select
  /VGRST         -- active-low clear for 3B
  AM7..AM0       -- vector-address lines from the Vector Address Selector
  STATE_CLK      -- state-machine latch clock from 2D pin 8

SIGNALS OUT:
  HALT*          -- clocked HALT, 3B pin 7
  F3..F0         -- latched PROM outputs, 3B pins 15,2,5,12
  O4..O1         -- direct PROM outputs, pins 9,10,11,12
  ST2            -- direct O3 net from PROM pin 10
  /LATCH0..3     -- active-low LS42 outputs 0-3
  /STROBE0..3    -- active-low LS42 outputs 4-7

CHIPS:
  1E    LS02     -- NOR for normal-mode PROM address bit A7
  2E    LS32     -- OR for Decoder Disable clear
  4C    LS157    -- PROM address bits A7-A4
  4A    LS157    -- PROM address bits A3-A0
  4B    82S129   -- 136021-109.4b, 256 x 4 bipolar PROM
  3B    LS174    -- state and HALT latch
  3D    LS42     -- active-low BCD-to-decimal decoder

PROM ADDRESS SELECTION:
  1E(LS02):
    NORMAL_A7 = NOR(GO, HALT*)                    -- pins 11,12 -> 13

  4C(LS157):
    S  = SA                                       -- pin 1
    /G = GND                                      -- pin 15

    channel 3: A(11)=NORMAL_A7, B(10)=AM7 -> Y(9)=PROM A7
    channel 4: A(14)=OP2,       B(13)=AM6 -> Y(12)=PROM A6
    channel 1: A(2)=OP1,        B(3)=AM5  -> Y(4)=PROM A5
    channel 2: A(5)=OP0,        B(6)=AM4  -> Y(7)=PROM A4

  4A(LS157):
    S  = SA                                       -- pin 1
    /G = GND                                      -- pin 15

    channel 1: A(2)=F3,  B(3)=AM3  -> Y(4)=PROM A3
    channel 2: A(5)=F2,  B(6)=AM2  -> Y(7)=PROM A2
    channel 3: A(11)=F1, B(10)=AM1 -> Y(9)=PROM A1
    channel 4: A(14)=F0, B(13)=AM0 -> Y(12)=PROM A0

  Selection:
    SA=0 -> {NORMAL_A7,OP2,OP1,OP0,F3,F2,F1,F0}
    SA=1 -> {AM7,AM6,AM5,AM4,AM3,AM2,AM1,AM0}

STATE PROM:
  4B(82S129):
    A0=pin 5, A1=pin 6, A2=pin 7, A3=pin 4
    A4=pin 3, A5=pin 2, A6=pin 1, A7=pin 15
    /CE1=pin 13, /CE2=pin 14 -> GND

    O1=pin 12
    O2=pin 11
    O3=pin 10 -> ST2 and 3D pin 13
    O4=pin 9  -> Decoder Disable and 3B pin 14

STATE LATCH:
  3B(LS174):
    /CLR = /VGRST                                -- pin 1
    CK   = STATE_CLK                             -- pin 9

    D pin 6  = HALT  -> Q pin 7  = HALT*
    D pin 14 = O4    -> Q pin 15 = F3
    D pin 3  = O3    -> Q pin 2  = F2
    D pin 4  = O2    -> Q pin 5  = F1
    D pin 13 = O1    -> Q pin 12 = F0

    D/Q pins 10/11 are unused.

STROBE AND LATCH DECODER:
  3D(LS42):
    A = O1                                      -- pin 15
    B = O2                                      -- pin 14
    C = O3                                      -- pin 13
    D = ST3                                     -- pin 12

    ST3=0 enables decoding of O3:O2:O1:
      000 -> pin 1 -> /LATCH0
      001 -> pin 2 -> /LATCH1
      010 -> pin 3 -> /LATCH2
      011 -> pin 4 -> /LATCH3
      100 -> pin 5 -> /STROBE0
      101 -> pin 6 -> /STROBE1
      110 -> pin 7 -> /STROBE2
      111 -> pin 9 -> /STROBE3

    ST3=1 makes outputs 0-7 inactive. LS42 outputs 8 and 9 are unused.

NOTES:
  - In normal mode the LS157 A inputs form the PROM address from the opcode,
    latched feedback F3..F0, and NORMAL_A7. In signature-analysis mode the B
    inputs apply vector-address lines AM7..AM0 directly to the PROM.
  - The LS157 enable pins are grounded. SA only drives their select pins.
  - ST2 is direct PROM output O3.
  - ST3 is the active-high output of the Decoder Disable timing circuit.
  - F3..F0 are the registered state feedback; O4..O1 are the current PROM
    result. The conventional $8-$F state names refer to {O4,O3,O2,O1}. O4
    releases Decoder Disable so that ST3 can go LOW; the LS42 then sees D=0
    and decodes O3:O2:O1 as outputs 0-7.


; ------------------------------------------------------------------------------
; BLOCK: State Machine Clock              *** VERIFIED 2026-07-16 ***
; Sheet 13A, PDF page 25
; ------------------------------------------------------------------------------

SIGNALS IN:
  ST2            -- direct PROM output O3
  /VMEM          -- active-low main-CPU vector-memory access request
  /1.5 MHZ       -- active-low 1.5 MHz timing signal
  12 MHZ         -- flip-flop clock
  P,R109         -- 1K pull-up to +5 V

SIGNALS OUT:
  STATE_CLK      -- 2D pin 8, clocks 3B pin 9
  CLOCK_D        -- 3E pin 10, also feeds Decoder Disable 2D pin 2

CHIPS:
  3E    S02      -- quad 2-input NOR; two gates used
  2D    LS74     -- upper half of dual D flip-flop

LOGIC:
  3E(S02):
    CLOCK_GATE = NOR(ST2, /VMEM)                  -- pins 5,6 -> 4
    CLOCK_D    = NOR(CLOCK_GATE, /1.5 MHZ)        -- pins 8,9 -> 10

    Pin 4 is wired directly to pin 8.

  2D(LS74), upper half:
    D    = CLOCK_D                                -- pin 12
    CK   = 12 MHZ                                 -- pin 11
    /PRE = P,R109                                 -- pin 10, pulled HIGH
    /CLR = P,R109                                 -- pin 13, pulled HIGH
    Q    = unused                                 -- pin 9
    /Q   = STATE_CLK                              -- pin 8

NOTES:
  - P,R109 holds both asynchronous controls HIGH.
  - CLOCK_D also feeds the D input of the Decoder Disable flip-flop.


; ------------------------------------------------------------------------------
; BLOCK: Decoder Disable                  *** VERIFIED 2026-07-16 ***
; Sheet 13A, PDF page 25
; ------------------------------------------------------------------------------

SIGNALS IN:
  SA             -- signature-analysis selection
  O4             -- direct PROM output, 4B pin 9
  CLOCK_D        -- State Machine Clock, 3E pin 10
  12 MHZ
  P,R109         -- 1K pull-up to +5 V

SIGNALS OUT:
  ST3            -- active-high LS42 decoder disable

CHIPS:
  2E    LS32     -- OR gate
  2D    LS74     -- lower half of dual D flip-flop

LOGIC:
  2E(LS32):
    DECODER_CLEAR_N = SA OR O4                    -- pins 13,12 -> 11

  2D(LS74), lower half:
    D    = CLOCK_D                                -- pin 2
    CK   = 12 MHZ                                 -- pin 3
    /PRE = P,R109                                 -- pin 4, pulled HIGH
    /CLR = DECODER_CLEAR_N                        -- pin 1
    Q    = unused                                 -- pin 5
    /Q   = ST3                                    -- pin 6

BEHAVIOR:
  SA=0 and O4=0:
    DECODER_CLEAR_N=0 -> Q=0 -> ST3=1 -> LS42 outputs 0-7 disabled

  O4=1 in normal operation:
    asynchronous clear is released; CLOCK_D is sampled at 12 MHz and ST3 can
    go LOW for the valid decode interval.

NOTES:
  - ST3 is the timing-qualified decoder-disable signal.
  - P,R109 only holds /PRE inactive.

; ------------------------------------------------------------------------------
; BLOCK: Vector Timer                     *** VERIFIED 2026-07-16 ***
; Sheet 13B, PDF page 26
; ------------------------------------------------------------------------------
;
; Four LS161 counters form an 8-bit SVEC timer or a 15-bit VCTR/CENTER timer.
; During NORM or SCALE the active-low parallel-load input performs a numerical
; right shift with HIGH bits entering from the top. During GO the counters
; count up; the terminal condition drives /STOP LOW.

SIGNALS IN:
  GO             -- count enable, VCTR OR CNTR
  NORM
  SCALE
  OP1
  /OP1
  /SA
  12 MHZ
  P,R111         -- 1K pull-up to +5 V

SIGNALS OUT:
  /STOP          -- active-low terminal-count signal

CHIPS:
  1D    LS161    -- bits 3:0
  1C    LS161    -- bits 7:4
  1B    LS161    -- bits 11:8
  2B    LS161    -- bits 14:12; QD is outside the 15-bit timer
  3H    LS02     -- NORM/SCALE load control
  1A    LS08     -- lower carry conjunction
  1E    LS02     -- OP1/bit-8 fill selection
  3A    LS00     -- two NAND gates used
  2C    LS11     -- three-input AND
  2A    LS20     -- dual four-input NAND; both gates used

COMMON LOAD CONTROL:
  3H(LS02):
    TIMER_LOAD_N = NOR(SCALE, NORM)                -- pins 11,12 -> 13

  TIMER_LOAD_N is connected to /LD pin 9 of all four LS161 counters:
    SCALE=0 and NORM=0 -> TIMER_LOAD_N=1 -> count/hold operation
    SCALE=1 or  NORM=1 -> TIMER_LOAD_N=0 -> synchronous parallel load

COUNTER 1D, BITS 3:0:
  1D(LS161):
    /CLR(1) = P,R111                               -- pulled HIGH
    CK(2)   = 12 MHZ
    A(3)    = QB(13)
    B(4)    = QC(12)
    C(5)    = QD(11)
    D(6)    = 1C QA(14)
    EP(7)   = GO
    /LD(9)  = TIMER_LOAD_N
    ET(10)  = GO

    QA,QB,QC,QD = bits 0,1,2,3
    RIP(15) -> 1C EP(7), 1A pin 2, 2C pin 9, 2A pin 9

COUNTER 1C, BITS 7:4:
  1C(LS161):
    /CLR(1) = P,R111                               -- pulled HIGH
    CK(2)   = 12 MHZ
    A(3)    = QB(13)
    B(4)    = QC(12)
    C(5)    = QD(11)
    D(6)    = 3A pin 6
    EP(7)   = 1D RIP(15)
    /LD(9)  = TIMER_LOAD_N
    ET(10)  = P,R111                               -- pulled HIGH

    QA,QB,QC,QD = bits 4,5,6,7
    QA(14)  -> 1D D(6)
    RIP(15) -> 1A pin 1, 2C pin 10, 2A pin 10

LOWER CARRY:
  1A(LS08):
    pin 1 = 1C RIP
    pin 2 = 1D RIP
    pin 3 = 1D_RIP AND 1C_RIP -> 1B EP(7)

COUNTER 1B, BITS 11:8:
  1B(LS161):
    /CLR(1) = /OP1
    CK(2)   = 12 MHZ
    A(3)    = QB(13)
    B(4)    = QC(12)
    C(5)    = QD(11)
    D(6)    = 2B QA(14)
    EP(7)   = 1A pin 3
    /LD(9)  = TIMER_LOAD_N
    ET(10)  = P,R111                               -- pulled HIGH

    QA,QB,QC,QD = bits 8,9,10,11
    QA(14)  -> 1E pin 3
    RIP(15) -> 2C pin 11, 2A pin 1

UPPER CARRY:
  2C(LS11):
    pin 11 = 1B RIP
    pin 10 = 1C RIP
    pin 9  = 1D RIP
    pin 8  = 1B_RIP AND 1C_RIP AND 1D_RIP
           -> 2B EP(7) and ET(10)

COUNTER 2B, BITS 14:12:
  2B(LS161):
    /CLR(1) = /OP1
    CK(2)   = 12 MHZ
    A(3)    = QB(13)
    B(4)    = QC(12)
    C(5)    = P,R111                               -- HIGH fill input
    D(6)    = P,R111                               -- HIGH; QD is unused
    EP(7)   = 2C pin 8
    /LD(9)  = TIMER_LOAD_N
    ET(10)  = 2C pin 8

    QA,QB,QC = bits 12,13,14
    QA(14) -> 1B D(6), 2A pin 2
    QB(13) -> 2A pin 4
    QC(12) -> 2A pin 5
    QD(11) is unused.

PARALLEL-LOAD FILL PATH:
  1E(LS02):
    FILL_SELECT_N = NOR(OP1, 1B_QA)                -- pins 2,3 -> 1

  3A(LS00), gate pins 4,5 -> 6:
    1C_D = NAND(/SA, FILL_SELECT_N)

  During VCTR or CENTER, OP1=0 and /SA=1, so 1C_D follows pre-load bit 8. The
  fixed HIGH inputs at 2B supply HIGH fill at bit 14 of the 15-bit timer.
  During SVEC, OP1=1 makes /OP1 LOW, holding 1B and 2B cleared, and forces
  1C_D HIGH to insert the HIGH fill bit at bit 7 of the 8-bit timer.

TERMINAL COUNT AND /STOP:
  2A(LS20), gate pins 1,2,4,5 -> 6:
    UPPER_N = NAND(1B_RIP, 2B_QA, 2B_QB, 2B_QC)

  3A(LS00), gate pins 9,10 -> 8:
    WIDTH_OK = NAND(UPPER_N, /OP1)

  2A(LS20), gate pins 9,10,12,13 -> 8:
    /STOP = NAND(1D_RIP, 1C_RIP, /SA, WIDTH_OK)

  Mode reduction:
    VCTR: OP1=0, /OP1=1
      WIDTH_OK = 1B_RIP AND 2B_QA AND 2B_QB AND 2B_QC
      /STOP goes LOW at the 15-bit terminal count.

    SVEC: OP1=1, /OP1=0
      1B and 2B are asynchronously cleared and WIDTH_OK=1.
      /STOP goes LOW at the 8-bit terminal count of 1D and 1C.

NOTES:
  - 2A is one LS20 package containing both four-input NAND gates.
  - The parallel connections A<-QB, B<-QC, C<-QD move each stored bit toward
    the least-significant end on every load, producing a numerical right shift.
  - /SA=0 keeps /STOP HIGH during signature analysis.
  - P,R111 is the R111 1K pull-up to +5 V shown on Sheet 11B.
